What to observe

  • IF — Instruction Fetch: fetch instruction from memory.
  • ID — Instruction Decode: decode and read registers.
  • EX — Execute: ALU operation or address calculation.
  • MEM — Memory Access: read or write data memory.
  • WB — Write Back: write result to a register.

Python simulation code

Save this code as pipeline_simulator.py and run with Python 3. The web page's interactive controls are implemented in JavaScript so this HTML file works directly in a browser.

# Pipeline Execution Simulator - COA Micro Project
STAGES = ["IF", "ID", "EX", "MEM", "WB"]

instructions = [
    {"name": "LOAD R1, [A]", "reads": [], "writes": "R1"},
    {"name": "ADD R2, R1, R3", "reads": ["R1"], "writes": "R2"},
    {"name": "SUB R4, R2, R5", "reads": ["R2"], "writes": "R4"},
    {"name": "AND R6, R4, R7", "reads": ["R4"], "writes": "R6"},
]

def find_raw_hazards(items):
    """Find read-after-write dependencies between instructions."""
    hazards = []
    for current in range(1, len(items)):
        for previous in range(current):
            written = items[previous]["writes"]
            if written and written in items[current]["reads"]:
                hazards.append(
                    (items[previous]["name"], items[current]["name"], written)
                )
    return hazards

def simulate(stage_count=5, forwarding=True):
    if stage_count not in (3, 4, 5):
        raise ValueError("stage_count must be 3, 4, or 5")

    stages = STAGES[:stage_count]
    print("Pipeline stages:", " -> ".join(stages))
    print("Forwarding:", "ON" if forwarding else "OFF")
    print("\\nInstruction dependencies:")
    for source, dependent, register in find_raw_hazards(instructions):
        print(f"RAW hazard on {register}: {source} -> {dependent}")

    # Ideal pipelined execution cycles = instructions + stages - 1.
    ideal_cycles = len(instructions) + len(stages) - 1
    # Simplified teaching estimate: each RAW dependency adds a stall
    # when forwarding is disabled. Real CPUs may use different rules.
    hazard_count = len(find_raw_hazards(instructions))
    stalls = 0 if forwarding else hazard_count
    total_cycles = ideal_cycles + stalls
    print("\\nInstruction count:", len(instructions))
    print("Ideal cycles:", ideal_cycles)
    print("Estimated stall cycles:", stalls)
    print("Estimated total cycles:", total_cycles)
    print("Done.")

if __name__ == "__main__":
    simulate(stage_count=5, forwarding=True)
